function startDailyLesson(){ const lesson=getDayLesson(); /* DAYS 1–30 — ALPHABET */ if(currentDay<=30){ showLesson( "Day "+currentDay+" — Alphabet", `
${lesson.letter}
${lesson.word}

${lesson.sound}

${lesson.sentence}

${ currentDay===30 ? '' : '' }
` ); return; } /* DAYS 31–60 — PHONICS */ if(currentDay>=31 && currentDay<=60){ const examples=lesson.examples || []; showLesson( "Day "+currentDay+" — Phonics", `
${lesson.letter}
${lesson.word}

${lesson.sound}

Today's Focus: ${lesson.focus}

Practice Words

${examples.join(" • ")}

${ currentDay===60 ? '' : '' }
` ); return; } /* DAYS 61–365 */ showLesson( "Day "+currentDay, `

📚 Future Curriculum Module

Day ${currentDay} is reserved for the next Class 1 English curriculum module.

This module will be activated as the 365-day curriculum is developed.

` ); }
